(use-package s)
(defmacro f-string (fmt)
  "Like `s-format' but with format fields in it.
FMT is a string to be expanded against the current lexical
environment. It is like what is used in `s-lex-format', but has
an expanded syntax to allow format-strings. For example:
${user-full-name 20s} will be expanded to the current value of
the variable `user-full-name' in a field 20 characters wide.
  (let ((f (sqrt 5)))  (f-string \"${f 1.2f}\"))
  will render as: 2.24
This function is inspired by the f-strings in Python 3.6, which I
enjoy using a lot."
  (let* ((matches (s-match-strings-all"${\\(?3:\\(?1:[^} ]+\\) *\\(?2:[^}]*\\)\\)}" fmt))
         (agetter (cl-loop for (m0 m1 m2 m3) in matches
                           collect `(cons ,m3  (format (format "%%%s" (if (string= ,m2 "")
                                                                          (if s-lex-value-as-lisp "S" "s")
                                                                       ,m2))
                                                      (symbol-value (intern ,m1)))))))
    `(s-format ,fmt 'aget (list ,@agetter))))

(defun get-authinfo-password (machine)
  "Read password from encrypted authinfo file."
  (let* ((auth-sources '("~/.authinfo"))
         (auth-info (auth-source-search :host machine :max 1)))
    (if auth-info
        (let ((secret (plist-get (car auth-info) :secret)))
          (if (functionp secret)
              (funcall secret)
            secret))
      (message "No matching auth info found for %s" machine))))

  (defun my/agent-pipe-default-wrap (content source-name query)
    "Default wrapper: tag CONTENT with SOURCE-NAME and optional QUERY."
    (concat (format "<from %s>\n" source-name)
            (when query (format "Query: %s\n\n" query))
            content
            (format "\n</from %s>" source-name)))

  (defcustom my/agent-pipe-wrap-format #'my/agent-pipe-default-wrap
    "Function called as (FN CONTENT SOURCE-NAME QUERY) to wrap piped content.
QUERY is nil in grab mode, the query string in ask/ask-fresh mode.
Return the wrapped string.  Set to nil to paste raw without wrapping."
    :type '(choice (const :tag "Raw, no wrapping" nil)
                   (function :tag "Wrapping function")))

Keys: :source :dest :query :subscription :submit :fresh-spawn-p :initialized.")

  (defun my/agent-pipe--dest-shell-buffer (buffer)
    "Return the underlying shell buffer for BUFFER, or nil if not a pipe dest."
    (with-current-buffer buffer
      (cond
       ((derived-mode-p 'agent-shell-mode) buffer)
       ((or (derived-mode-p 'agent-shell-viewport-view-mode)
            (derived-mode-p 'agent-shell-viewport-edit-mode))
        (agent-shell-viewport--shell-buffer buffer)))))

  (defun my/agent-pipe--ensure-dest (buffer)
    "Signal `user-error' unless BUFFER is a valid pipe destination."
    (unless (my/agent-pipe--dest-shell-buffer buffer)
      (user-error "Not an agent-shell or viewport buffer")))

  (defun my/agent-pipe--candidates (dest-buffer)
    "Live `agent-shell' buffers, excluding the shell behind DEST-BUFFER."
    (let ((self (my/agent-pipe--dest-shell-buffer dest-buffer)))
      (seq-remove (lambda (b) (eq b self))
                  (seq-filter #'buffer-live-p (agent-shell-buffers)))))

  (defun my/agent-pipe--pick-source (dest-buffer &optional reprompt)
    "Return a source `agent-shell' buffer for DEST-BUFFER.
With REPROMPT non-nil, ignore the sticky cache."
    (let* ((cached (and (not reprompt)
                        (buffer-local-value 'my/agent-pipe--sticky-source
                                            dest-buffer)))
           (sticky (and cached (buffer-live-p cached) cached))
           (candidates (my/agent-pipe--candidates dest-buffer))
           (chosen
            (cond
             (sticky sticky)
             ((null candidates)
              (user-error "No other live agent-shell buffers"))
             ((= 1 (length candidates)) (car candidates))
             (t (get-buffer
                 (completing-read
                  "Pipe source: "
                  (mapcar #'buffer-name candidates) nil t))))))
      (with-current-buffer dest-buffer
        (setq my/agent-pipe--sticky-source chosen))
      chosen))

  (cl-defun my/agent-pipe--inject (source-buffer dest-buffer content
                                                 &key query submit)
    "Inject CONTENT from SOURCE-BUFFER into DEST-BUFFER.
QUERY is non-nil for ask/ask-fresh modes.  SUBMIT auto-sends in the dest."
    (let ((wrapped (if my/agent-pipe-wrap-format
                       (funcall my/agent-pipe-wrap-format
                                content (buffer-name source-buffer) query)
                     content)))
      (with-current-buffer dest-buffer
        (cond
         ((derived-mode-p 'agent-shell-mode)
          (agent-shell--insert-to-shell-buffer
           :shell-buffer dest-buffer
           :text wrapped
           :submit submit
           :no-focus t))
         ((or (derived-mode-p 'agent-shell-viewport-view-mode)
              (derived-mode-p 'agent-shell-viewport-edit-mode))
          (when submit
            (message
             "agent-pipe: :submit ignored (viewport destination unsupported)"))
          (agent-shell-viewport--show-buffer
           :append wrapped
           :shell-buffer (agent-shell-viewport--shell-buffer dest-buffer)))
         (t (user-error "Pipe destination is not an agent-shell or viewport"))))))

  (defun my/agent-pipe--dest-has-inflight-p (dest-buffer)
    "Return non-nil if any in-flight pipe targets DEST-BUFFER."
    (seq-some (lambda (p) (eq dest-buffer (plist-get p :dest)))
              my/agent-pipe--inflight))

  (defun my/agent-pipe--drop-entry (entry)
    "Remove ENTRY from `my/agent-pipe--inflight'."
    (setq my/agent-pipe--inflight (delq entry my/agent-pipe--inflight)))

  (defun my/agent-pipe--make-handler (dest-buffer)
    "Return a `prompt-ready' closure for pipes targeting DEST-BUFFER."
    (lambda (_event)
      (when-let* ((entry (seq-find
                          (lambda (p) (eq dest-buffer (plist-get p :dest)))
                          my/agent-pipe--inflight)))
        (let ((source (plist-get entry :source))
              (query  (plist-get entry :query))
              (submit (plist-get entry :submit))
              (fresh  (plist-get entry :fresh-spawn-p))
              (sub    (plist-get entry :subscription)))
          (cond
           ((or (not (buffer-live-p source))
                (not (buffer-live-p dest-buffer)))
            (when (and sub (buffer-live-p source))
              (with-current-buffer source
                (agent-shell-unsubscribe :subscription sub)))
            (my/agent-pipe--drop-entry entry))
           ((not (plist-get entry :initialized))
            (plist-put entry :initialized t)
            (with-current-buffer source
              (agent-shell--insert-to-shell-buffer
               :shell-buffer source
               :text query
               :submit t
               :no-focus t))
            (message "agent-pipe: query sent to %s, waiting..."
                     (buffer-name source)))
           (t
            (with-current-buffer source
              (agent-shell-unsubscribe :subscription sub))
            (let ((response (with-current-buffer source
                              (shell-maker-last-output))))
              (my/agent-pipe--drop-entry entry)
              (if (and response
                       (not (string-empty-p (string-trim response))))
                  (progn
                    (my/agent-pipe--inject source dest-buffer response
                                           :query query :submit submit)
                    (when (and fresh
                               my/agent-pipe-kill-fresh-source-after
                               (buffer-live-p source))
                      (kill-buffer source))
                    (message "agent-pipe: %s -> %s done"
                             (buffer-name source)
                             (buffer-name dest-buffer)))
                (message "agent-pipe: %s returned empty response"
                         (buffer-name source))))))))))

  (defun my/agent-pipe--start-ask (source dest query submit fresh-spawn-p)
    "Subscribe SOURCE for a two-phase ask flow into DEST."
    (when (my/agent-pipe--dest-has-inflight-p dest)
      (user-error "Pipe already in flight for this destination"))
    (let* ((handler (my/agent-pipe--make-handler dest))
           (entry (list :source source :dest dest :query query
                        :subscription nil :submit submit
                        :fresh-spawn-p fresh-spawn-p :initialized nil)))
      (push entry my/agent-pipe--inflight)
      (let ((token (agent-shell-subscribe-to
                    :shell-buffer source
                    :event 'prompt-ready
                    :on-event handler)))
        (plist-put entry :subscription token))
      (message "agent-pipe: %s -> %s initializing..."
               (buffer-name source) (buffer-name dest))))

  (defun my/agent-pipe-grab ()
    "Grab last output (or active region) from a source agent-shell.
With `C-u' re-pick the source.  With `C-u C-u' also submit in the dest."
    (interactive)
    (my/agent-pipe--ensure-dest (current-buffer))
    (let* ((dest (current-buffer))
           (reprompt (equal current-prefix-arg '(4)))
           (submit (equal current-prefix-arg '(16)))
           (source (my/agent-pipe--pick-source dest reprompt))
           (content (with-current-buffer source
                      (if (use-region-p)
                          (buffer-substring-no-properties
                           (region-beginning) (region-end))
                        (shell-maker-last-output)))))
      (when (or (null content)
                (string-empty-p (string-trim (or content ""))))
        (user-error "Source %s has no content to grab" (buffer-name source)))
      (my/agent-pipe--inject source dest content :submit submit)))

  (defun my/agent-pipe-ask ()
    "Send a query to a source agent-shell; stage its answer in this buffer.
With `C-u' re-pick the source.  With `C-u C-u' also submit in the dest."
    (interactive)
    (my/agent-pipe--ensure-dest (current-buffer))
    (let* ((dest (current-buffer))
           (reprompt (equal current-prefix-arg '(4)))
           (submit (equal current-prefix-arg '(16)))
           (source (my/agent-pipe--pick-source dest reprompt))
           (query (if (use-region-p)
                      (buffer-substring-no-properties
                       (region-beginning) (region-end))
                    (read-string (format "Query for %s: "
                                         (buffer-name source))))))
      (when (string-empty-p (string-trim query))
        (user-error "Empty query"))
      (my/agent-pipe--start-ask source dest query submit nil)))

  (defun my/agent-pipe-ask-fresh ()
    "Spawn a fresh source agent-shell, ask QUERY, stage its answer here.
With `C-u' pick the config builder.  With `C-u C-u' also submit in the dest."
    (interactive)
    (my/agent-pipe--ensure-dest (current-buffer))
    (let* ((dest (current-buffer))
           (pick-config (equal current-prefix-arg '(4)))
           (submit (equal current-prefix-arg '(16)))
           (builder (if pick-config
                        (intern (completing-read
                                 "Config builder: "
                                 (mapcar #'symbol-name
                                         my/agent-pipe-config-builders)
                                 nil t))
                      my/agent-pipe-default-config))
           (query (if (use-region-p)
                      (buffer-substring-no-properties
                       (region-beginning) (region-end))
                    (read-string "Query for fresh source: "))))
      (when (string-empty-p (string-trim query))
        (user-error "Empty query"))
      (let ((source (agent-shell--start
                     :config (funcall builder)
                     :no-focus t
                     :new-session t
                     :session-strategy 'new)))
        (unless source
          (user-error "Failed to spawn fresh source"))
        (my/agent-pipe--start-ask source dest query submit t))))

  (defun my/agent-pipe-cancel ()
    "Cancel in-flight pipes targeting this buffer.
With `C-u' cancel all in-flight pipes globally."
    (interactive)
    (let* ((all (equal current-prefix-arg '(4)))
           (dest (current-buffer))
           (targets (if all
                        (copy-sequence my/agent-pipe--inflight)
                      (seq-filter
                       (lambda (p) (eq dest (plist-get p :dest)))
                       my/agent-pipe--inflight))))
      (if (null targets)
          (message "No pipes to cancel.")
        (dolist (entry targets)
          (let ((source (plist-get entry :source))
                (sub (plist-get entry :subscription)))
            (when (and sub (buffer-live-p source))
              (with-current-buffer source
                (agent-shell-unsubscribe :subscription sub))))
          (my/agent-pipe--drop-entry entry))
        (message "Cancelled %d pipe(s)." (length targets)))))
